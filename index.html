function applyAutoRigging() { if (!activeModel) { showToast("Model Load Karein!"); closeRigOverlay(); return; } showToast("Binding Full Skeleton to Model..."); // Existing helpers aur handles clear karein if (skeletonHelper) scene.remove(skeletonHelper); controlHandles.forEach(h => scene.remove(h)); controlHandles = []; const container = document.getElementById('joints-container'); const nodes = container.querySelectorAll('.joint-node'); const rect = container.getBoundingClientRect(); // Screen coordinates to 3D positions calculate karne ke liye Raycaster const raycaster = new THREE.Raycaster(); const mouse = new THREE.Vector2(); const jointWorldPositions = {}; nodes.forEach(node => { const id = node.dataset.id; const nodeRect = node.getBoundingClientRect(); const x = nodeRect.left + nodeRect.width / 2; const y = nodeRect.top + nodeRect.height / 2; mouse.x = ((x - rect.left) / rect.width) * 2 - 1; mouse.y = -((y - rect.top) / rect.height) * 2 + 1; raycaster.setFromCamera(mouse, camera); // Model mesh par intersect check karein const intersects = raycaster.intersectObject(activeModel, true); if (intersects.length > 0) { jointWorldPositions[id] = intersects[0].point.clone(); } else { // Agar direct model par raycast fail ho to camera plane me project karein const targetVec = new THREE.Vector3(); raycaster.ray.at(1.5, targetVec); jointWorldPositions[id] = targetVec; } }); closeRigOverlay(); // Sabhi 13 Joints ke liye Bones Create Karein const boneMap = {}; defaultJoints.forEach(j => { const bone = new THREE.Bone(); bone.name = j.id; if (jointWorldPositions[j.id]) { bone.position.copy(jointWorldPositions[j.id]); } boneMap[j.id] = bone; }); // Skeleton Hierarchy Structure Bind Karein const rootBone = boneMap['hips'] || new THREE.Bone(); if (boneMap['hips'] && boneMap['spine']) boneMap['hips'].add(boneMap['spine']); if (boneMap['spine'] && boneMap['neck']) boneMap['spine'].add(boneMap['neck']); if (boneMap['neck'] && boneMap['head']) boneMap['neck'].add(boneMap['head']); // Right Arm if (boneMap['spine'] && boneMap['shoulder_r']) boneMap['spine'].add(boneMap['shoulder_r']); if (boneMap['shoulder_r'] && boneMap['elbow_r']) boneMap['shoulder_r'].add(boneMap['elbow_r']); if (boneMap['elbow_r'] && boneMap['wrist_r']) boneMap['elbow_r'].add(boneMap['wrist_r']); // Left Arm if (boneMap['spine'] && boneMap['shoulder_l']) boneMap['spine'].add(boneMap['shoulder_l']); if (boneMap['shoulder_l'] && boneMap['elbow_l']) boneMap['shoulder_l'].add(boneMap['elbow_l']); if (boneMap['elbow_l'] && boneMap['wrist_l']) boneMap['elbow_l'].add(boneMap['wrist_l']); // Right Leg if (boneMap['hips'] && boneMap['knee_r']) boneMap['hips'].add(boneMap['knee_r']); if (boneMap['knee_r'] && boneMap['ankle_r']) boneMap['knee_r'].add(boneMap['ankle_r']); // Left Leg if (boneMap['hips'] && boneMap['knee_l']) boneMap['hips'].add(boneMap['knee_l']); if (boneMap['knee_l'] && boneMap['ankle_l']) boneMap['knee_l'].add(boneMap['ankle_l']); scene.add(rootBone); // Har Bone joint par 3D Interactive Handles banao Object.keys(boneMap).forEach(key => { const b = boneMap[key]; const dot = new THREE.Mesh( new THREE.SphereGeometry(0.04, 12, 12), new THREE.MeshBasicMaterial({ color: 0x007aff, depthTest: false }) ); dot.userData.targetBone = b; scene.add(dot); controlHandles.push(dot); }); // Skeleton Connections Line Helper skeletonHelper = new THREE.SkeletonHelper(rootBone); skeletonHelper.material.linewidth = 2; scene.add(skeletonHelper); // Transform Gizmo ko Hips/Spine par attach karein transformControls.attach(rootBone); showToast("Full Rigging Successfully Applied!"); }
