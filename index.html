function applyBoneRig() { if (!selectedObj) { showMsg('Pehle model select karein!'); return; } // Purane Rig ya Helper ko remove karein if (currentSkeletonGroup) { scene.remove(currentSkeletonGroup); currentSkeletonGroup = null; } // 1. Agar model ke andar pehle se Skinned Bones majood hain: let hasSkeleton = false; selectedObj.traverse((child) => { if (child.isBone || child.isSkinnedMesh) hasSkeleton = true; }); if (hasSkeleton) { const helper = new THREE.SkeletonHelper(selectedObj); helper.material.linewidth = 3; helper.material.depthTest = false; // Always visible on top scene.add(helper); currentSkeletonGroup = helper; showMsg('Real Skeleton Loaded!'); return; } // 2. Agar Custom Auto-Rig banana hai to Model ka Bounding Box (Size & Height) nikalein: const box = new THREE.Box3().setFromObject(selectedObj); const size = new THREE.Vector3(); const center = new THREE.Vector3(); box.getSize(size); box.getCenter(center); const modelHeight = size.y || 1.8; const minY = box.min.y; const rigGroup = new THREE.Group(); const mat = new THREE.MeshBasicMaterial({ color: 0x34c759, wireframe: true, depthTest: false, transparent: true, opacity: 0.8 }); const jointMat = new THREE.MeshBasicMaterial({ color: 0xff9500, depthTest: false }); const createBone = (startY, endY, radiusTop = 0.03, radiusBot = 0.03) => { const h = endY - startY; const geo = new THREE.CylinderGeometry(radiusTop, radiusBot, h, 8); geo.translate(0, h / 2, 0); const mesh = new THREE.Mesh(geo, mat); mesh.position.y = startY; return mesh; }; const createJoint = (y, r = 0.04) => { const j = new THREE.Mesh(new THREE.SphereGeometry(r, 8, 8), jointMat); j.position.y = y; return j; }; // Humanoid proportions (Feet se Head tak automatically stretch hoga): const spineStart = minY + modelHeight * 0.5; const chestHeight = minY + modelHeight * 0.75; const neckHeight = minY + modelHeight * 0.85; const headHeight = minY + modelHeight * 0.95; // Spine & Head rigGroup.add(createBone(spineStart, chestHeight, 0.04, 0.05)); rigGroup.add(createBone(chestHeight, neckHeight, 0.03, 0.04)); rigGroup.add(createJoint(chestHeight)); const headMesh = new THREE.Mesh(new THREE.SphereGeometry(modelHeight * 0.08, 12, 12), mat); headMesh.position.y = headHeight; rigGroup.add(headMesh); // Legs (Left & Right) const hipWidth = size.x * 0.15 || 0.15; [-hipWidth, hipWidth].forEach(x => { const hipY = spineStart; const kneeY = minY + modelHeight * 0.25; const footY = minY; const thigh = createBone(kneeY, hipY, 0.03, 0.04); thigh.position.x = x; const shin = createBone(footY, kneeY, 0.02, 0.03); shin.position.x = x; rigGroup.add(thigh); rigGroup.add(shin); const kneeJoint = createJoint(kneeY, 0.035); kneeJoint.position.x = x; rigGroup.add(kneeJoint); }); // Arms (Left & Right) const shoulderWidth = size.x * 0.25 || 0.25; [-shoulderWidth, shoulderWidth].forEach(x => { const shoulderY = chestHeight; const elbowY = minY + modelHeight * 0.62; const handY = minY + modelHeight * 0.5; const arm = createBone(elbowY, shoulderY, 0.025, 0.03); arm.position.x = x; const forearm = createBone(handY, elbowY, 0.02, 0.025); forearm.position.x = x; rigGroup.add(arm); rigGroup.add(forearm); }); scene.add(rigGroup); currentSkeletonGroup = rigGroup; showMsg('Auto-Scaled Rig Applied!'); }
