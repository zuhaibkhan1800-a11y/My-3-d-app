function applyAutoRigging() { if (!activeModel) { showToast("Model Load Karein!"); closeRigOverlay(); return; } showToast("Binding Rig directly to Model..."); if (skeletonHelper) scene.remove(skeletonHelper); controlHandles.forEach(h => scene.remove(h)); controlHandles = []; const container = document.getElementById('joints-container'); const nodes = container.querySelectorAll('.joint-node'); const rect = container.getBoundingClientRect(); const raycaster = new THREE.Raycaster(); const mouse = new THREE.Vector2(); const jointWorldPositions = {}; // 1. Raycast Screen Coordinates to 3D Space nodes.forEach(node => { const id = node.dataset.id; const nodeRect = node.getBoundingClientRect(); const x = nodeRect.left + nodeRect.width / 2; const y = nodeRect.top + nodeRect.height / 2; mouse.x = ((x - rect.left) / rect.width) * 2 - 1; mouse.y = -((y - rect.top) / rect.height) * 2 + 1; raycaster.setFromCamera(mouse, camera); const intersects = raycaster.intersectObject(activeModel, true); if (intersects.length > 0) { jointWorldPositions[id] = intersects[0].point.clone(); } else { const targetVec = new THREE.Vector3(); raycaster.ray.at(2.0, targetVec); jointWorldPositions[id] = targetVec; } }); closeRigOverlay(); const hierarchy = { 'hips': null, 'spine': 'hips', 'neck': 'spine', 'head': 'neck', 'shoulder_r': 'spine', 'elbow_r': 'shoulder_r', 'wrist_r': 'elbow_r', 'shoulder_l': 'spine', 'elbow_l': 'shoulder_l', 'wrist_l': 'elbow_l', 'knee_r': 'hips', 'ankle_r': 'knee_r', 'knee_l': 'hips', 'ankle_l': 'knee_l' }; const bones = {}; defaultJoints.forEach(j => { const b = new THREE.Bone(); b.name = j.id; bones[j.id] = b; }); Object.keys(hierarchy).forEach(id => { const parentId = hierarchy[id]; const bone = bones[id]; const worldPos = jointWorldPositions[id] || new THREE.Vector3(0, 1, 0); if (parentId && bones[parentId]) { const parentBone = bones[parentId]; parentBone.add(bone); const parentWorldPos = jointWorldPositions[parentId] || new THREE.Vector3(0, 1, 0); bone.position.subVectors(worldPos, parentWorldPos); } else { bone.position.copy(worldPos); } }); const rootBone = bones['hips']; scene.add(rootBone); rootBone.updateMatrixWorld(true); // 2. Visible Joint Dots (Always Rendered On Top) Object.keys(bones).forEach(key => { const b = bones[key]; const dotGeom = new THREE.SphereGeometry(0.04, 16, 16); const dotMat = new THREE.MeshBasicMaterial({ color: 0x007aff, depthTest: false, depthWrite: false }); const dot = new THREE.Mesh(dotGeom, dotMat); dot.renderOrder = 999; dot.userData.targetBone = b; scene.add(dot); controlHandles.push(dot); }); // 3. Visible Skeleton Helper Lines skeletonHelper = new THREE.SkeletonHelper(rootBone); skeletonHelper.material.linewidth = 3; skeletonHelper.material.depthTest = false; skeletonHelper.material.depthWrite = false; skeletonHelper.renderOrder = 998; scene.add(skeletonHelper); transformControls.attach(rootBone); showToast("Rig Perfectly Fitted & Visible!"); }
