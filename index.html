function applyAutoRigging() { if (!activeModel) { showToast("Model Load Karein!"); closeRigOverlay(); return; } showToast("Binding Rig directly to Model..."); if (skeletonHelper) scene.remove(skeletonHelper); controlHandles.forEach(h => scene.remove(h)); controlHandles = []; const container = document.getElementById('joints-container'); const nodes = container.querySelectorAll('.joint-node'); const rect = container.getBoundingClientRect(); const raycaster = new THREE.Raycaster(); const mouse = new THREE.Vector2(); const jointWorldPositions = {}; // 1. Screen coordinates se Model par EXACT 3D World Positions nikalein nodes.forEach(node => { const id = node.dataset.id; const nodeRect = node.getBoundingClientRect(); const x = nodeRect.left + nodeRect.width / 2; const y = nodeRect.top + nodeRect.height / 2; mouse.x = ((x - rect.left) / rect.width) * 2 - 1; mouse.y = -((y - rect.top) / rect.height) * 2 + 1; raycaster.setFromCamera(mouse, camera); const intersects = raycaster.intersectObject(activeModel, true); if (intersects.length > 0) { jointWorldPositions[id] = intersects[0].point.clone(); } else { const targetVec = new THREE.Vector3(); raycaster.ray.at(2.0, targetVec); jointWorldPositions[id] = targetVec; } }); closeRigOverlay(); // 2. Bone Hierarchy Tree Define Karein const hierarchy = { 'hips': null, 'spine': 'hips', 'neck': 'spine', 'head': 'neck', 'shoulder_r': 'spine', 'elbow_r': 'shoulder_r', 'wrist_r': 'elbow_r', 'shoulder_l': 'spine', 'elbow_l': 'shoulder_l', 'wrist_l': 'elbow_l', 'knee_r': 'hips', 'ankle_r': 'knee_r', 'knee_l': 'hips', 'ankle_l': 'knee_l' }; const bones = {}; // 3. Sabhi Bones create karein defaultJoints.forEach(j => { const b = new THREE.Bone(); b.name = j.id; bones[j.id] = b; }); // 4. Parents aur Children ko sahi LOCAL positions ke saath connect karein Object.keys(hierarchy).forEach(id => { const parentId = hierarchy[id]; const bone = bones[id]; const worldPos = jointWorldPositions[id] || new THREE.Vector3(0, 1, 0); if (parentId && bones[parentId]) { const parentBone = bones[parentId]; parentBone.add(bone); // Parent ke relative LOCAL position calculate karein const parentWorldPos = jointWorldPositions[parentId] || new THREE.Vector3(0, 1, 0); bone.position.subVectors(worldPos, parentWorldPos); } else { // Hips (Root) ki World Position set karein bone.position.copy(worldPos); } }); const rootBone = bones['hips']; scene.add(rootBone); rootBone.updateMatrixWorld(true); // 5. Visual Controls & Skeleton Lines Object.keys(bones).forEach(key => { const b = bones[key]; const dot = new THREE.Mesh( new THREE.SphereGeometry(0.03, 12, 12), new THREE.MeshBasicMaterial({ color: 0x007aff, depthTest: false }) ); dot.userData.targetBone = b; scene.add(dot); controlHandles.push(dot); }); skeletonHelper = new THREE.SkeletonHelper(rootBone); skeletonHelper.material.linewidth = 3; scene.add(skeletonHelper); transformControls.attach(rootBone); showToast("Rig Perfectly Fitted to Model Body!"); }
