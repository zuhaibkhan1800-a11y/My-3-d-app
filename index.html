function startRenderLoop() { requestAnimationFrame(startRenderLoop); orbitControls.update(); // 1. Check if user is actively dragging the TransformControls const isDragging = transformControls && transformControls.dragging; if (!isDragging) { // 2. Snap handle to bone ONLY when NOT dragging to avoid fighting the drag for (let key in ikTargetsMap) { let boneName = (key === 'r_arm') ? 'wrist_r' : (key === 'l_arm') ? 'wrist_l' : (key === 'r_leg') ? 'ankle_r' : 'ankle_l'; let bone = bonesMap[boneName]; let targetObj = ikTargetsMap[key]; if (bone && targetObj && targetObj !== transformControls.object) { characterGroup.updateMatrixWorld(true); let boneWorldPos = new THREE.Vector3(); bone.getWorldPosition(boneWorldPos); // 3. Update target world position precisely targetObj.position.copy(boneWorldPos); } } } else { // 4. When dragging, drive the IK solver from the active target handle's world position const activeObj = transformControls.object; if (activeObj && activeObj.userData.isIKTarget) { const chain = activeObj.userData.chain; // [root, mid, end] solveTargetIK(chain[0], chain[1], chain[2], activeObj); } } if (activeSkeleton) { activeSkeleton.update(); } renderer.render(scene, camera); drawTimelineCanvas(); }
